+class BookmarkNotifier extends StateNotifier<AsyncValue<void>> {
+  final Ref ref;
+
+  BookmarkNotifier(this.ref) : super(const AsyncValue.data(null));
+
+  Future<void> quickSave(String url) async {
+    state = const AsyncValue.loading();
+    try {
+      final repo = ref.read(repositoryProvider);
+      
+      // Attempt to fetch metadata
+      String? title;
+      String? description;
+      String? imageUrl;
+      String? sourceDomain;
+
+      try {
+        final uri = Uri.parse(url);
+        sourceDomain = uri.host;
+        var data = await MetadataFetch.extract(url);
+        if (data != null) {
+          title = data.title;
+          description = data.description;
+          imageUrl = data.image;
+        }
+      } catch (e) {
+        // Fallback or ignore
+        print('Error fetching metadata: $e');
+      }
+
+      final bookmark = Bookmark()
+        ..url = url
+        ..title = title ?? url
+        ..description = description
+        ..imageUrl = imageUrl
+        ..sourceDomain = sourceDomain
+        ..createdAt = DateTime.now()
+        ..isRead = false;
+
+      await repo.addBookmark(bookmark);
+      
+      // Invalidate the provider to refresh the UI
+      ref.invalidate(bookmarksProvider);
+      
+      state = const AsyncValue.data(null);
+    } catch (e, st) {
+      state = AsyncValue.error(e, st);
+    }
+  }
+
+  Future<void> markAsRead(Bookmark bookmark) async {
+    final repo = ref.read(repositoryProvider);
+    bookmark.isRead = true;
+    await repo.updateBookmark(bookmark);
+    ref.invalidate(bookmarksProvider);
+    if (bookmark.collection.value != null) {
+      ref.invalidate(collectionBookmarksProvider(bookmark.collection.value!.id));
+    }
+  }
+
+  Future<void> deleteBookmark(Bookmark bookmark) async {
+    final repo = ref.read(repositoryProvider);
+    await repo.deleteBookmark(bookmark.id);
+    ref.invalidate(bookmarksProvider);
+    if (bookmark.collection.value != null) {
+      ref.invalidate(collectionBookmarksProvider(bookmark.collection.value!.id));
+    }
+  }

+class NeoButton extends StatefulWidget {
+  final VoidCallback onPressed;
+  final Widget child;
+  final Color color;
+  final double borderWidth;
+  final double borderRadius;
+  final double shadowOffset;
+  final EdgeInsetsGeometry padding;
+
+  const NeoButton({
+    Key? key,
+    required this.onPressed,
+    required this.child,
+    this.color = AppTheme.primary,
+    this.borderWidth = 3.0,
+    this.borderRadius = 12.0,
+    this.shadowOffset = 5.0,
+    this.padding = const EdgeInsets.symmetric(horizontal: 24, vertical: 14),
+  }) : super(key: key);
+
+  @override
+  State<NeoButton> createState() => _NeoButtonState();
+}
+
+class _NeoButtonState extends State<NeoButton> {
+  bool _isPressed = false;
+
+  void _handleTapDown(TapDownDetails details) {
+    HapticFeedback.lightImpact();
+    setState(() {
+      _isPressed = true;
+    });
+  }
+
+  void _handleTapUp(TapUpDetails details) {
+    setState(() {
+      _isPressed = false;
+    });
+    widget.onPressed();
+  }
+
+  void _handleTapCancel() {
+    setState(() {
+      _isPressed = false;
+    });
+  }
+
+  @override
+  Widget build(BuildContext context) {
+    return GestureDetector(
+      onTapDown: _handleTapDown,
+      onTapUp: _handleTapUp,
+      onTapCancel: _handleTapCancel,
+      child: Transform.translate(
+        offset: Offset(
+          _isPressed ? widget.shadowOffset : 0,
+          _isPressed ? widget.shadowOffset : 0,
+        ),
+        child: AnimatedContainer(
+          duration: const Duration(milliseconds: 100),
+          padding: widget.padding,
+          decoration: BoxDecoration(
+            color: widget.color,
+            borderRadius: BorderRadius.circular(widget.borderRadius),
+            border: Border.all(
+              color: AppTheme.border,
+              width: widget.borderWidth,
+            ),
+            boxShadow: _isPressed
+                ? []
+                : [
+                    BoxShadow(
+                      color: AppTheme.border,
+                      offset: Offset(widget.shadowOffset, widget.shadowOffset),
+                      blurRadius: 0,
+                      spreadRadius: 0,
+                    ),
+                  ],
+          ),
+          child: widget.child,
+        ),
+      ),
+    );
+  }
