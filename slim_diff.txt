+  Future<void> quickSave(String url) async {

+    try {
+      final repo = ref.read(repositoryProvider);
+      
+      // Attempt to fetch metadata
+      String? title;
+      String? description;
+      String? imageUrl;
+      String? sourceDomain;

+        // Fallback or ignore
+        print('Error fetching metadata: $e');
+      }
+
+      final bookmark = Bookmark()
+        ..url = url
+        ..title = title ?? url
+        ..description = description
+        ..imageUrl = imageUrl
+        ..sourceDomain = sourceDomain
+        ..createdAt = DateTime.now()
+        ..isRead = false;
+
+      await repo.addBookmark(bookmark);
+      
+      // Invalidate the provider to refresh the UI
+      ref.invalidate(bookmarksProvider);

+import 'package:isar/isar.dart';
+import 'package:path_provider/path_provider.dart';
+import '../models/bookmark.dart';
+import '../models/collection.dart';
+
+class IsarRepository {
+  late Future<Isar> db;
+
+  IsarRepository() {
+    db = openDB();
+  }
+
+  Future<Isar> openDB() async {
+    if (Isar.instanceNames.isEmpty) {
+      final dir = await getApplicationDocumentsDirectory();
+      return await Isar.open(
+        [BookmarkSchema, BookmarkCollectionSchema],
+        directory: dir.path,
+        inspector: true,
+      );
+    }
+    return Future.value(Isar.getInstance());
+  }
+
+  // --- Collection Methods ---
+  Future<List<BookmarkCollection>> getCollections() async {
+    final isar = await db;
+    return await isar.bookmarkCollections.where().findAll();
+  }
+
+  Future<void> addCollection(BookmarkCollection collection) async {
+    final isar = await db;
+    await isar.writeTxn(() async {
+      await isar.bookmarkCollections.put(collection);
+    });
+  }
+
+  // --- Bookmark Methods ---
+  Future<List<Bookmark>> getBookmarks() async {
+    final isar = await db;
+    return await isar.bookmarks.where().sortByCreatedAtDesc().findAll();
+  }
+
+  Future<List<Bookmark>> getBookmarksByCollection(Id collectionId) async {
+    final isar = await db;
+    // Assuming the link has been setup properly
+    // This fetches bookmarks linking to the collection
+    return await isar.bookmarks
+        .filter()
+        .collection((q) => q.idEqualTo(collectionId))
+        .sortByCreatedAtDesc()
+        .findAll();
+  }
+
+  Future<void> addBookmark(Bookmark bookmark) async {
+    final isar = await db;
+    await isar.writeTxn(() async {
+      await isar.bookmarks.put(bookmark);
+      await bookmark.collection.save();
+    });
+  }
+
+  Future<void> updateBookmark(Bookmark bookmark) async {
+    final isar = await db;
+    await isar.writeTxn(() async {
+      await isar.bookmarks.put(bookmark);
+    });
+  }
+  
+  Future<void> deleteBookmark(Id id) async {
+    final isar = await db;
+    await isar.writeTxn(() async {
+      await isar.bookmarks.delete(id);
+    });
+  }

+class NeoButton extends StatefulWidget {
+  final VoidCallback onPressed;
+  final Widget child;
+  final Color color;
+  final double borderWidth;
+  final double borderRadius;
+  final double shadowOffset;
+  final EdgeInsetsGeometry padding;
+
+  const NeoButton({
+    Key? key,
+    required this.onPressed,
+    required this.child,
+    this.color = AppTheme.primary,
+    this.borderWidth = 3.0,
+    this.borderRadius = 12.0,
+    this.shadowOffset = 5.0,
+    this.padding = const EdgeInsets.symmetric(horizontal: 24, vertical: 14),
+  }) : super(key: key);
+
+  @override
+  State<NeoButton> createState() => _NeoButtonState();
+}
+
+class _NeoButtonState extends State<NeoButton> {
+  bool _isPressed = false;
+
+  void _handleTapDown(TapDownDetails details) {
+    HapticFeedback.lightImpact();
+    setState(() {
+      _isPressed = true;
+    });
+  }
+
+  void _handleTapUp(TapUpDetails details) {
+    setState(() {
+      _isPressed = false;
+    });
+    widget.onPressed();
+  }
+
+  void _handleTapCancel() {
+    setState(() {
+      _isPressed = false;
+    });
+  }
+
+  @override
+  Widget build(BuildContext context) {
+    return GestureDetector(
+      onTapDown: _handleTapDown,
+      onTapUp: _handleTapUp,
+      onTapCancel: _handleTapCancel,
+      child: Transform.translate(
+        offset: Offset(
+          _isPressed ? widget.shadowOffset : 0,
+          _isPressed ? widget.shadowOffset : 0,
+        ),
+        child: AnimatedContainer(
+          duration: const Duration(milliseconds: 100),
+          padding: widget.padding,
+          decoration: BoxDecoration(
+            color: widget.color,
+            borderRadius: BorderRadius.circular(widget.borderRadius),
+            border: Border.all(
+              color: AppTheme.border,
+              width: widget.borderWidth,
+            ),
+            boxShadow: _isPressed
+                ? []
+                : [
+                    BoxShadow(
+                      color: AppTheme.border,
+                      offset: Offset(widget.shadowOffset, widget.shadowOffset),
+                      blurRadius: 0,
+                      spreadRadius: 0,
+                    ),
+                  ],
+          ),
+          child: widget.child,
+        ),
+      ),
+    );
+  }
